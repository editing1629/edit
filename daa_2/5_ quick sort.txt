
import random
import time

# Deterministic Quick Sort
def deterministic(arr):
    if len(arr) <= 1:
        return arr

    pivot = arr[0]
    left = [x for x in arr[1:] if x <= pivot]
    right = [x for x in arr[1:] if x > pivot]

    return deterministic(left) + [pivot] + deterministic(right)


# Randomized Quick Sort
def randomized(arr):
    if len(arr) <= 1:
        return arr

    pivot = random.choice(arr)
    left = [x for x in arr if x < pivot]
    right = [x for x in arr if x > pivot]
    equal = [x for x in arr if x == pivot]

    return randomized(left) + equal + randomized(right)


# Accept input
n = int(input("Enter number of elements: "))
arr = [int(input("Enter element: ")) for i in range(n)]

# Deterministic Quick Sort
start = time.time()
a = deterministic(arr.copy())
t1 = time.time() - start

print("\nDeterministic Quick Sort:", a)
print("Execution Time:", t1, "seconds")

# Randomized Quick Sort
start = time.time()
b = randomized(arr.copy())
t2 = time.time() - start

print("\nRandomized Quick Sort:", b)
print("Execution Time:", t2, "seconds")

# Complexity Analysis
print("\nTime Complexity:")
print("Deterministic: Best/Average O(n log n), Worst O(n^2)")
print("Randomized: Expected O(n log n), Worst O(n^2)")
